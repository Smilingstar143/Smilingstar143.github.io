2. Program to implement (DDA) Line Drawing Algorithm

#include<graphics.h>
#include<conio.h>
#include<math.h>
#include<iostream.h>

void dda(float x1,float y1,float x2,float y2)
{
    float dx,dy,steps;
    float xinc,yinc;
    float x,y;
    int i;

    dx=x2-x1;
    dy=y2-y1;

    if((fabs(dx)>fabs(dy)))
        steps=fabs(dx);
    else
        steps=fabs(dy);

    xinc=dx/steps;
    yinc=dy/steps;

    x=x1;
    y=y1;

    putpixel((int)(x+0.5),(int)(y+0.5),WHITE);

    for(i=0; i<steps; i++)
    {
        x=x+xinc;
        y=y+yinc;
        putpixel((int)(x+0.5),(int)(y+0.5),WHITE);
    }
}

int main()
{
    int gd=DETECT,gm;
    float x1,y1,x2,y2;

    initgraph(&gd,&gm,"C:\\turboc3\\bgi");

    cout<<"Enter x1,y1:";
    cin>>x1>>y1;

    cout<<"Enter x2,y2:";
    cin>>x2>>y2;

    dda(x1,y1,x2,y2);

    outtextxy(0,200,(char*)"DDA line drawing algorithm");

    getch();
    closegraph();

    return 0;
}

3. Program to implement Bresenham's Line Drawing Algorithm

#include <graphics.h>
#include <stdio.h>
#include <conio.h>
#include <math.h>
#include <iostream.h>

void bresenham(int x1, int y1, int x2, int y2)
{
    int dx, dy, sx, sy, x, y, p;

    dx = abs(x2 - x1);
    dy = abs(y2 - y1);

    sx = (x2 > x1) ? 1 : -1;
    sy = (y2 > y1) ? 1 : -1;

    x = x1;
    y = y1;

    if (dx > dy)
    {
        p = 2 * dy - dx;
        for (int i = 0; i <= dx; i++)
        {
            putpixel(x, y, WHITE);
            x += sx;
            if (p >= 0)
            {
                y += sy;
                p -= 2 * dx;
            }
            p += 2 * dy;
        }
    }
    else
    {
        p = 2 * dx - dy;
        for (int i = 0; i <= dy; i++)
        {
            putpixel(x, y, WHITE);
            y += sy;
            if (p >= 0)
            {
                x += sx;
                p -= 2 * dy;
            }
            p += 2 * dx;
        }
    }
}

void main()
{
    int gd = DETECT, gm;
    int x1, y1, x2, y2;

    cout<<"Enter starting point (x1 y1)";
    cin>>x1>>y1;
    cout<<"Enter ending point (x2 y2)";
    cin>>x2>>y2;

    initgraph(&gd, &gm, "C:\\Turboc3\\BGI");

    bresenham(x1, y1, x2, y2);

    getch();
    closegraph();
}

4. Program to draw a circle using Bresenham's Circle Algorithm

#include <graphics.h>
#include <iostream.h>
#include <conio.h>

void drawCircle(int xc, int yc, int x, int y)
{
    putpixel(xc + x, yc + y, WHITE);
    putpixel(xc - x, yc + y, WHITE);
    putpixel(xc + x, yc - y, WHITE);
    putpixel(xc - x, yc - y, WHITE);
    putpixel(xc + y, yc + x, WHITE);
    putpixel(xc - y, yc + x, WHITE);
    putpixel(xc + y, yc - x, WHITE);
    putpixel(xc - y, yc - x, WHITE);
}

int main()
{
    int gd = DETECT, gm;
    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    int xc, yc, r;

    cout << "Enter center (xc yc): ";
    cin >> xc >> yc;

    cout << "Enter radius: ";
    cin >> r;

    int x = 0;
    int y = r;
    int p = 3 - 2 * r;

    while (x <= y)
    {
        drawCircle(xc, yc, x, y);

        if (p < 0)
            p = p + 4 * x + 6;
        else
        {
            p = p + 4 * (x - y) + 10;
            y--;
        }

        x++;
    }

    outtextxy(50, 50, "Bresenham's Circle Drawing Algorithm");

    getch();
    closegraph();

    return 0;
}

5. Program to perform 2D Transformations

#include <graphics.h>
#include <iostream.h>
#include <math.h>
#include <conio.h>

int main()
{
    int gd = DETECT, gm;
    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    int x1, y1, x2, y2, x3, y3;
    int ch;

    cout << "Enter Triangle Coordinates:\n";

    cout << "x1 y1: ";
    cin >> x1 >> y1;

    cout << "x2 y2: ";
    cin >> x2 >> y2;

    cout << "x3 y3: ";
    cin >> x3 >> y3;

    // Draw Original Triangle
    line(x1, y1, x2, y2);
    line(x2, y2, x3, y3);
    line(x3, y3, x1, y1);

    cout << "\n2D Transformations";
    cout << "\n1. Translation";
    cout << "\n2. Scaling";
    cout << "\n3. Rotation";
    cout << "\nEnter your choice: ";
    cin >> ch;

    switch(ch)
    {
        case 1:
        {
            int tx, ty;

            cout << "Enter Translation Factors (tx ty): ";
            cin >> tx >> ty;

            x1 += tx; y1 += ty;
            x2 += tx; y2 += ty;
            x3 += tx; y3 += ty;

            break;
        }

        case 2:
        {
            float sx, sy;

            cout << "Enter Scaling Factors (sx sy): ";
            cin >> sx >> sy;

            x1 = x1 * sx;
            y1 = y1 * sy;
            x2 = x2 * sx;
            y2 = y2 * sy;
            x3 = x3 * sx;
            y3 = y3 * sy;

            break;
        }

        case 3:
        {
            float angle;

            cout << "Enter Rotation Angle (degrees): ";
            cin >> angle;

            float rad = angle * 3.14159 / 180;

            int tx1 = x1*cos(rad) - y1*sin(rad);
            int ty1 = x1*sin(rad) + y1*cos(rad);

            int tx2 = x2*cos(rad) - y2*sin(rad);
            int ty2 = x2*sin(rad) + y2*cos(rad);

            int tx3 = x3*cos(rad) - y3*sin(rad);
            int ty3 = x3*sin(rad) + y3*cos(rad);

            x1 = tx1; y1 = ty1;
            x2 = tx2; y2 = ty2;
            x3 = tx3; y3 = ty3;

            break;
        }

        default:
            cout << "Invalid Choice!";
    }

    // Draw Transformed Triangle
    setcolor(RED);

    line(x1, y1, x2, y2);
    line(x2, y2, x3, y3);
    line(x3, y3, x1, y1);

    outtextxy(150, 150, "2D Transformation");

    getch();
    closegraph();

    return 0;
}

6. Program for Line Clipping using Cohen-Sutherland Algorithm

#include <graphics.h>
#include <iostream.h>
#include <conio.h>

const int INSIDE = 0;
const int LEFT = 1;
const int RIGHT = 2;
const int BOTTOM = 4;
const int TOP = 8;

int xmin = 100, ymin = 100, xmax = 300, ymax = 300;

int computeCode(double x, double y)
{
    int code = INSIDE;

    if (x < xmin)
        code |= LEFT;
    else if (x > xmax)
        code |= RIGHT;

    if (y < ymin)
        code |= BOTTOM;
    else if (y > ymax)
        code |= TOP;

    return code;
}

int main()
{
    int gd = DETECT, gm;
    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    double x1, y1, x2, y2;

    cout << "Enter x1 y1: ";
    cin >> x1 >> y1;

    cout << "Enter x2 y2: ";
    cin >> x2 >> y2;

    rectangle(xmin, ymin, xmax, ymax);

    line(x1, y1, x2, y2);

    int code1 = computeCode(x1, y1);
    int code2 = computeCode(x2, y2);

    int accept = 0;

    while (1)
    {
        if ((code1 == 0) && (code2 == 0))
        {
            accept = 1;
            break;
        }
        else if (code1 & code2)
        {
            break;
        }
        else
        {
            int codeOut;
            double x, y;

            if (code1 != 0)
                codeOut = code1;
            else
                codeOut = code2;

            if (codeOut & TOP)
            {
                x = x1 + (x2 - x1) * (ymax - y1) / (y2 - y1);
                y = ymax;
            }
            else if (codeOut & BOTTOM)
            {
                x = x1 + (x2 - x1) * (ymin - y1) / (y2 - y1);
                y = ymin;
            }
            else if (codeOut & RIGHT)
            {
                y = y1 + (y2 - y1) * (xmax - x1) / (x2 - x1);
                x = xmax;
            }
            else
            {
                y = y1 + (y2 - y1) * (xmin - x1) / (x2 - x1);
                x = xmin;
            }

            if (codeOut == code1)
            {
                x1 = x;
                y1 = y;
                code1 = computeCode(x1, y1);
            }
            else
            {
                x2 = x;
                y2 = y;
                code2 = computeCode(x2, y2);
            }
        }
    }

    if (accept)
    {
        setcolor(RED);
        line(x1, y1, x2, y2);
        outtextxy(50, 50, "Clipped Line");
    }
    else
    {
        outtextxy(10, 10, "Line Rejected");
    }

    getch();
    closegraph();

    return 0;
}

8. Program to perform 3D Transformations

#include <graphics.h>
#include <iostream.h>
#include <math.h>
#include <conio.h>

int main()
{
    int gd = DETECT, gm;
    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    int x[8] = {100, 200, 200, 100, 150, 250, 250, 150};
    int y[8] = {100, 100, 200, 200, 50, 50, 150, 150};
    int z[8] = {0, 0, 0, 0, 100, 100, 100, 100};

    setcolor(WHITE);

    // Front face
    line(x[0], y[0], x[1], y[1]);
    line(x[1], y[1], x[2], y[2]);
    line(x[2], y[2], x[3], y[3]);
    line(x[3], y[3], x[0], y[0]);

    // Back face
    line(x[4], y[4], x[5], y[5]);
    line(x[5], y[5], x[6], y[6]);
    line(x[6], y[6], x[7], y[7]);
    line(x[7], y[7], x[4], y[4]);

    // Connecting edges
    line(x[0], y[0], x[4], y[4]);
    line(x[1], y[1], x[5], y[5]);
    line(x[2], y[2], x[6], y[6]);
    line(x[3], y[3], x[7], y[7]);

    int choice;

    cout << "\n3D Transformations";
    cout << "\n1. Translation";
    cout << "\n2. Scaling";
    cout << "\n3. Rotation (Z-axis)";
    cout << "\nEnter your choice: ";
    cin >> choice;

    switch(choice)
    {
        case 1:
        {
            int tx, ty, tz;

            cout << "Enter Translation Factors (tx ty tz): ";
            cin >> tx >> ty >> tz;

            for(int i = 0; i < 8; i++)
            {
                x[i] += tx;
                y[i] += ty;
                z[i] += tz;
            }

            break;
        }

        case 2:
        {
            float sx, sy, sz;

            cout << "Enter Scaling Factors (sx sy sz): ";
            cin >> sx >> sy >> sz;

            for(int i = 0; i < 8; i++)
            {
                x[i] *= sx;
                y[i] *= sy;
                z[i] *= sz;
            }

            break;
        }

        case 3:
        {
            float angle;

            cout << "Enter Rotation Angle: ";
            cin >> angle;

            float rad = angle * 3.14159 / 180;

            for(int i = 0; i < 8; i++)
            {
                int xt = x[i];
                int yt = y[i];

                x[i] = xt * cos(rad) - yt * sin(rad);
                y[i] = xt * sin(rad) + yt * cos(rad);
            }

            break;
        }

        default:
            cout << "Invalid Choice!";
            getch();
            closegraph();
            return 0;
    }

    setcolor(RED);

    // Front face
    line(x[0], y[0], x[1], y[1]);
    line(x[1], y[1], x[2], y[2]);
    line(x[2], y[2], x[3], y[3]);
    line(x[3], y[3], x[0], y[0]);

    // Back face
    line(x[4], y[4], x[5], y[5]);
    line(x[5], y[5], x[6], y[6]);
    line(x[6], y[6], x[7], y[7]);
    line(x[7], y[7], x[4], y[4]);

    // Connecting edges
    line(x[0], y[0], x[4], y[4]);
    line(x[1], y[1], x[5], y[5]);
    line(x[2], y[2], x[6], y[6]);
    line(x[3], y[3], x[7], y[7]);

    outtextxy(10, 10, "3D Transformation");

    getch();
    closegraph();

    return 0;
}
